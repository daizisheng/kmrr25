\section{Unrestricted sums \texorpdfstring{$B+B$}{B+B} at the lower density threshold}\label{IA:sec:u32a}

In this section we answer Question~\S3.2(a) of~\cite{kra2023problems}, negatively, under each of
its two possible readings. Throughout this part of the paper $B+B=\{b+b':b,b'\in B\}$ denotes the
\emph{unrestricted} sumset, which contains the diagonal elements $2b$, and
$\N_0=\N\cup\{0\}$.

Kousek and Radi\'c~\cite[Theorem~1.2]{kousek2024infinite} proved that a set $A\subset\N$ contains
$B+B$ for some infinite $B\subset\N$ as soon as $\udens(A)>5/6$ or $\ldens(A)>3/4$, and showed
that both thresholds are sharp~\cite[Propositions~4.1 and~4.2]{kousek2024infinite}. Right after
reporting this, the survey asks the following.

\begin{question}{\S3.2(a)}\label{IA:q:u32a}
What occurs if we only assume, for example, that $|A\cap[N]|>\tfrac34N+\log N$ for infinitely many
$N$? That is: must such a set $A\subset\N$ contain $B+B$ for some infinite set $B\subset\N$?
\end{question}

Read literally (``for infinitely many $N$''), the hypothesis is implied by $\udens(A)>3/4$, so it
lies far below the upper density threshold $5/6$. The question is presumably meant as a boundary case
of the lower density threshold, that is, with ``for all sufficiently large $N$''. We answer both
readings.

\begin{theorem}\label{IA:thm:u32a}
\begin{enumerate}[label=\textup{(\alph*)}]
\item\label{IA:it:u32a-io} There is a set $A\subset\N$ such that for every $C>0$ we have
$|A\cap[N]|>\tfrac34N+C\log N$ for infinitely many $N$, and such that $B+B\not\subset A$ for every
infinite $B\subset\N$.
\item\label{IA:it:u32a-all} There are a set $A\subset\N$ and a non-decreasing unbounded function
$\psi\colon\N\to[0,\infty)$ such that
\[
|A\cap[N]|\ \ge\ \tfrac34N+\psi(N)\log N\qquad\text{for all sufficiently large }N,
\]
and $B+B\not\subset A$ for every infinite $B\subset\N$.
\end{enumerate}
\end{theorem}

Since $\psi\to\infty$, part~\ref{IA:it:u32a-all} gives $|A\cap[N]|>\frac34N+C\log N$ for all large
$N$, for every constant $C$ and every base of the logarithm.

\begin{remark}\label{IA:rem:u32a}
\begin{enumerate}[label=(\roman*)]
\item Part~\ref{IA:it:u32a-io} is essentially contained in~\cite{kousek2024infinite}: the set $A'$
of~\cite[Proposition~4.1]{kousek2024infinite} has upper density $5/6>3/4$ and contains no $B+B$,
so it satisfies the literal hypothesis of \cref{IA:q:u32a}. We give an explicit variant with an exact
count, to make the size of the excess visible. The literal reading is therefore answered by the
known sharpness examples, and the natural question is the one in part~\ref{IA:it:u32a-all}.
\item The set in part~\ref{IA:it:u32a-all} has lower density exactly $3/4$: at least $3/4$ by
construction, and at most $3/4$ by~\cite[Theorem~1.2]{kousek2024infinite}, since it contains no
$B+B$. The proof uses the diagonal elements $2b\in B+B$ in an
essential way (see Step~(iv) in the proof of \cref{IA:prop:rot}).
\item We do not know which functions $g(N)=o(N)$ have the property that $|A\cap[N]|\ge \frac34N+g(N)$
for all large $N$ forces $B+B\subset A$ for some infinite $B$; part~\ref{IA:it:u32a-all} shows that
$g(N)=\psi(N)\log N$ does not, for a suitable $\psi\to\infty$.
\end{enumerate}
\end{remark}

\subsection{The literal reading}

\begin{proof}[Proof of \cref{IA:thm:u32a}\ref{IA:it:u32a-io}]
Let $S=\bigcup_{n\ge1}\big[4^n,\,2\cdot4^n-2n\big)\cap\Z$ and $A=(2\N-1)\cup(S\cap2\N)$.

\emph{$S$ contains no $D+D$ with $D$ infinite.} Suppose $D+D\subset S$, let $d'=\min D$ and let
$d\in D$ be large. Since $2d\in S$, there is $m$ with $4^m\le 2d<2\cdot4^m-2m$, and $m\to\infty$ as
$d\to\infty$; take $d$ so large that $m\ge d'$. Then $2\cdot 4^{m-1}\le d\le d+d'<4^m-m+d'\le 4^m$.
But $S\cap[2\cdot4^{m-1},4^m)=\varnothing$, because the block of index $m-1$ ends below
$2\cdot4^{m-1}$ and the block of index $m$ starts at $4^m$. This contradicts $d+d'\in S$.

\emph{$A$ contains no $B+B$ with $B$ infinite.} If $B+B\subset A$, choose an infinite $B'\subset B$
whose elements all have the same parity. Then $B'+B'$ consists of even numbers, so
$B'+B'\subset A\cap2\N\subset S$, which is impossible.

\emph{Counting.} Let $N_n=2\cdot4^n-2n-1$. The odd numbers in $[N_n]$ number $(N_n+1)/2=4^n-n$. The
block $[4^m,2\cdot4^m-2m)$ has even length and starts at an even number, so it contains
$4^m/2-m$ even numbers; the blocks with $m\le n$ lie in $[N_n]$ and the block with $m=n+1$ does not
meet it. Hence
\[
|A\cap[N_n]|=4^n-n+\sum_{m=1}^n\Big(\frac{4^m}2-m\Big)=4^n-n+\frac{4^{n+1}-4}{6}-\frac{n(n+1)}2,
\]
and subtracting $\frac34N_n=\frac32 4^n-\frac32n-\frac34$ gives
\[
|A\cap[N_n]|-\tfrac34N_n=\frac{4^n}{6}-\frac{n^2}{2}+\frac1{12}.
\]
This exceeds $C\log N_n=O(n)$ for every fixed $C$ once $n$ is large (for $C=1$ and the natural
logarithm already for $n\ge3$).
\end{proof}

\subsection{Rotation blocks}
For the second reading we first reduce to a statement about a set $E$ of density $1/2$. For
$E\subset\N$ write $2E=\{2e:e\in E\}$.

\begin{lemma}\label{IA:lem:parity}
Let $E\subset\N$ and $A=(2\N-1)\cup2E$. If $B+B\subset A$ for some infinite $B\subset\N$, then there
are an infinite $D\subset\N_0$ and $t\in\{0,1\}$ with $D+D+t\subset E$. Moreover
$|A\cap[N]|=\lceil N/2\rceil+|E\cap[1,\lfloor N/2\rfloor]|$ for every $N\in\N$.
\end{lemma}

\begin{proof}
Choose an infinite $B'\subset B$ of constant parity; then $B'+B'\subset A\cap2\N=2E$. If $B'\subset
2\N$, put $D=\{b/2:b\in B'\}$, so that $2(D+D)=B'+B'$ and $D+D\subset E$. If $B'$ consists of odd
numbers, put $D=\{(b-1)/2:b\in B'\}\subset\N_0$, so that $B'+B'=2(D+D+1)$ and $D+D+1\subset E$. The
counting formula is clear.
\end{proof}

Let $\rho(x)=\one_{[0,1/2)}(\{x\})-\frac12$ for $x\in\R$, so that $\rho$ is $1$-periodic,
right-continuous, has jumps exactly at $\frac12\Z$, and satisfies $\rho(x+\frac12)=-\rho(x)$. For
$n\in\N$ let $\lambda(n)=\lfloor\log_2n\rfloor$ and $\varepsilon(n)=(-1)^{\lambda(n)}$. Note that
\begin{equation}\label{IA:eq:lambda}
\lambda(2n+t)=\lambda(n)+1\qquad(n\in\N,\ t\in\{0,1\}).
\end{equation}

\begin{definition}\label{IA:def:rot}
Let $\alpha,\beta\in\R$ and let $(w_i)_{i\ge1}$ be positive integers. Let
$J_0=\{x\in\T:\{x-\beta\}\in[0,\tfrac12)\}$ and $J_1=\T\setminus J_0$, and put
\[
R=\{n\in\N:\ n\alpha\bmod 1\in J_{\lambda(n)\bmod 2}\},\qquad
G=\bigcup_{i\ge1}\,[2^i-w_i,\,2^i+w_i),\qquad E=R\setminus G .
\]
\end{definition}

So $R$ is a Bohr set on each dyadic block, using the half-circle $J_0$ on blocks
$[2^j,2^{j+1})$ with $j$ even and the complementary half-circle on blocks with $j$ odd; $E$ is
obtained by removing slowly growing neighbourhoods of the powers of~$2$. For $\alpha\in\{0,\frac12\}$
these are block sets of the kind used in~\cite[Section~4]{kousek2024infinite}.

\begin{proposition}\label{IA:prop:rot}
In \cref{IA:def:rot}, assume $w_i\to\infty$. Then there are no infinite $D\subset\N_0$ and
$t\in\{0,1\}$ with $D+D+t\subset E$.
\end{proposition}

\begin{proof}
Suppose $D$ is infinite, $t\in\{0,1\}$ and $D+D+t\subset E$; removing $0$ from $D$ we may assume
$D\subset\N$.
\begin{enumerate}[label=(\roman*),leftmargin=*]
\item For $d\in D$ put $z_d=d\alpha+t\alpha/2\bmod1$. Then $(d+d'+t)\alpha\equiv z_d+z_{d'}$ and
$(2d+t)\alpha\equiv2z_d\pmod 1$.
\item For $k=0,1,2,3$ let $Q_k=\{z\in\T:\{z-\beta/2\}\in[\frac k4,\frac{k+1}4)\}$. If $z,z'\in
Q_k$, write $z\equiv\beta/2+x$, $z'\equiv\beta/2+x'$ with $x,x'\in[\frac k4,\frac{k+1}4)$; then
$z+z'\equiv\beta+(x+x')$ with $x+x'\in[\frac k2,\frac{k+1}2)$, so $z+z'\in J_{k\bmod2}$.
In particular $2z\in J_{k\bmod 2}$ as well.
\item By the pigeonhole principle there are $\sigma\in\{0,1\}$, $k\in\{0,1,2,3\}$ and an infinite
$D'\subset D$ with $\lambda(d)\equiv\sigma\pmod 2$ and $z_d\in Q_k$ for all $d\in D'$.
\item Let $d\in D'$. Since $2d+t\in E\subset R$ and $\lambda(2d+t)=\lambda(d)+1$
by~\eqref{IA:eq:lambda}, we have $(2d+t)\alpha\in J_{1-\sigma}$. By (i) and (ii),
$(2d+t)\alpha\equiv2z_d\in J_{k\bmod 2}$. As $J_0\cap J_1=\varnothing$, this forces $k\equiv1-\sigma
\pmod 2$. (This is the only place where a diagonal element is used.)
\item Let $d_0=\min D'$ and let $d'\in D'$, $d'>d_0$. Then $n:=d_0+d'+t\in E\subset R$ and, by (i)
and (ii), $n\alpha\equiv z_{d_0}+z_{d'}\in J_{k\bmod2}=J_{1-\sigma}$. Since $n\in R$, this means
$\lambda(n)\equiv1-\sigma\not\equiv\lambda(d')\pmod 2$. As $n>d'$, some power $2^i$ lies in
$(d',n]$. Since $n\notin G$ and $n\ge 2^i$, we get $n\ge2^i+w_i$, so
$w_i\le n-2^i<n-d'=d_0+t$.
\item Letting $d'\to\infty$ through $D'$, the index $i$ in (v) tends to infinity (because $2^i>d'$),
while $w_i<d_0+t$ stays bounded. This contradicts $w_i\to\infty$.\qedhere
\end{enumerate}
\end{proof}

\begin{lemma}\label{IA:lem:excess}
In \cref{IA:def:rot}, for every $M\in\N$,
\[
|E\cap[1,M)|\ \ge\ \frac{M-1}{2}+S_{\alpha,\beta}(M)-\sum_{i\ge1:\ 2^i-w_i<M}2w_i,
\]
where $S_{\alpha,\beta}(M)=\sum_{1\le n<M}\varepsilon(n)\,\rho(n\alpha-\beta)$.
\end{lemma}

\begin{proof}
If $\lambda(n)$ is even, then $\one_R(n)=\one_{[0,1/2)}(\{n\alpha-\beta\})$, so
$\one_R(n)-\frac12=\rho(n\alpha-\beta)$; if $\lambda(n)$ is odd, then
$\one_R(n)-\frac12=-\rho(n\alpha-\beta)$. Thus $\one_R(n)-\frac12=\varepsilon(n)\rho(n\alpha-\beta)$
and $|R\cap[1,M)|=\frac{M-1}2+S_{\alpha,\beta}(M)$. Finally $|G\cap[1,M)|$ is at most the sum of the
lengths $2w_i$ of the intervals $[2^i-w_i,2^i+w_i)$ that meet $[1,M)$.
\end{proof}

Thus everything hinges on making the twisted discrepancy $S_{\alpha,\beta}(M)$ grow faster than
$\log M$ for \emph{all} large $M$: the gaps cost $\sum 2w_i$, which grows faster than $\log M$
because $w_i\to\infty$.

\begin{proposition}\label{IA:prop:TD}
There are an irrational $\alpha$ and integers $1\le N_1\le N_2\le\cdots$ such that
\[
S_{\alpha,1/6}(M)\ \ge\ i\log M\qquad\text{for all } i\ge1 \text{ and all } M\ge N_i .
\]
\end{proposition}

We prove \cref{IA:prop:TD} in \cref{IA:ssec:TD}, and first deduce part~\ref{IA:it:u32a-all} of the
theorem.

\begin{proof}[Proof of \cref{IA:thm:u32a}\ref{IA:it:u32a-all}]
Let $\alpha$ and $(N_i)$ be as in \cref{IA:prop:TD}; we may assume $N_1\ge2$. Since
$|S_\alpha(M)|\le(M-1)/2$, for each $M\ge2$ the inequality $S_\alpha(M)\ge i\log M$ fails for large
$i$, so $N_i\to\infty$. Let $\psi_0(M)=\max\{i:N_i\le M\}$ for $M\ge N_1$ and $\psi_0(M)=0$ for
$M<N_1$. Then $\psi_0$ is non-decreasing and unbounded, and
$S_{\alpha,1/6}(M)\ge\psi_0(M)\log M$ for $M\ge N_1$. Take $\beta=\frac16$ and
\[
w_i=\max\Big(1,\ \min\big(\lfloor 2^i/4\rfloor,\ \lfloor\psi_0(2^{i-1})/16\rfloor\big)\Big)\qquad(i\ge1),
\]
so that $w_i\to\infty$, and let $E$ be as in \cref{IA:def:rot}.

We bound the gap term of \cref{IA:lem:excess}. We have $w_1=1$. For $i\ge2$ we have $w_i\le2^i/4$, so
$2^i-w_i<M$ implies $\frac34 2^i<M$, hence $2^{i-1}<M$, hence $\psi_0(2^{i-1})\le\psi_0(M)$ and
$w_i\le\max(1,\psi_0(M)/16)$. The indices $i\ge 2$ with $\frac342^i<M$ satisfy
$i<\log_2M+1$, so there are at most $\log_2M+2$ indices in the sum. Hence, once $\psi_0(M)\ge16$,
\[
\sum_{2^i-w_i<M}2w_i\le\frac{\psi_0(M)}{8}\,(\log_2M+2)=\psi_0(M)\Big(\frac{\log M}{8\log 2}+\frac14\Big)
\le\psi_0(M)\big(0.19\log M+0.25\big).
\]
By \cref{IA:lem:excess}, for all large $M$,
\[
|E\cap[1,M)|-\frac{M-1}2\ \ge\ \psi_0(M)\big(0.81\log M-0.25\big)\ \ge\ \tfrac12\psi_0(M)\log M+1 .
\]
Let $A=(2\N-1)\cup2E$. By \cref{IA:lem:parity}, with $M=\lfloor N/2\rfloor+1$,
\[
|A\cap[N]|=\lceil N/2\rceil+|E\cap[1,M)|\ \ge\ \frac N2+\frac{N-1}4+\tfrac12\psi_0(M)\log M+1
\ \ge\ \frac34N+\tfrac12\psi_0(M)\log(N/2)
\]
for all large $N$. With $\psi(N)=\frac14\psi_0(\lfloor N/2\rfloor+1)$, which is non-decreasing and
unbounded, we get $|A\cap[N]|\ge\frac34N+\psi(N)\log N$ for all large $N$. Finally, if $B+B\subset A$
with $B$ infinite, \cref{IA:lem:parity} produces an infinite $D\subset\N_0$ and $t\in\{0,1\}$ with
$D+D+t\subset E$, contradicting \cref{IA:prop:rot}.
\end{proof}


\subsection{A twisted discrepancy estimate}\label{IA:ssec:TD}
For $\theta\in\R$ and $n\in\Z$ let $r_\theta(n)=\rho(n\theta-\frac16)$ and
$S_\theta(M)=\sum_{1\le n<M}\varepsilon(n)r_\theta(n)$, so that $S_\theta=S_{\theta,1/6}$. Let
$\mathcal Q$ be the set of rationals $\theta=a/q$ with
\[
q=3\cdot2^p\ (p\ge1),\qquad 0<a<q,\qquad\gcd(a,q)=1 .
\]
For $\theta=a/q\in\mathcal Q$ (always written in lowest terms) put
\[
\Delta(\theta)=\sum_{2^p\le n<2^{p+1}}r_\theta(n),\qquad G(\theta)=(-1)^p\Delta(\theta),\qquad
\mathcal D(\theta)=\sup_{u\le v}\Big|\sum_{u\le n<v}r_\theta(n)\Big| .
\]
Note that $q$ is even and $a$ is odd, and that $2^p=q/3$.

\begin{lemma}\label{IA:lem:TD0}
Let $\theta=a/q\in\mathcal Q$.
\begin{enumerate}[label=\textup{(\roman*)}]
\item $\sum_{n=u}^{u+q-1}r_\theta(n)=0$ for every $u\in\Z$; in particular $\mathcal D(\theta)\le q/2$.
\item For every $j\ge p$, $\sum_{2^j\le n<2^{j+1}}\varepsilon(n)r_\theta(n)=G(\theta)$.
\item $S_\theta(2^j)=S_\theta(2^p)+(j-p)G(\theta)$ for $j\ge p$, and
$|S_\theta(M)-S_\theta(2^j)|\le\mathcal D(\theta)$ whenever $2^j\le M\le 2^{j+1}$.
\end{enumerate}
\end{lemma}

\begin{proof}
(i) We have $n\theta-\frac16=(na-2^{p-1})/q$ with $2^{p-1}\in\Z$. As $n$ runs through $q$
consecutive integers, $na-2^{p-1}$ runs through all residues modulo $q$, and $\rho(k/q)=\frac12$ for
exactly $q/2$ residues $k$ and $-\frac12$ for the others. Hence the sum vanishes, the function
$D(N)=\sum_{0\le n<N}r_\theta(n)$ is $q$-periodic, every interval sum equals a sum over fewer than $q$
consecutive integers, and $\mathcal D(\theta)\le q/2$.

(ii) For $j\ge p$ let $u_j\in\{1,2\}$ with $u_j\equiv2^{j-p}\pmod 3$; thus $u_j=1$ if $j-p$ is even
and $u_j=2$ otherwise. Since $q=3\cdot2^p$ divides $2^p(2^{j-p}-u_j)$, periodicity gives
$D(2^j)=D(2^pu_j)$. Therefore
$\sum_{2^j\le n<2^{j+1}}r_\theta(n)=D(2^pu_{j+1})-D(2^pu_j)=(-1)^{j-p}\big(D(2^{p+1})-D(2^p)\big)
=(-1)^{j-p}\Delta(\theta)$. On $[2^j,2^{j+1})$ we have $\varepsilon=(-1)^j$, which gives (ii).

(iii) The first claim follows from (ii). If $2^j\le M\le2^{j+1}$, then
\[
S_\theta(M)-S_\theta(2^j)=(-1)^j\sum_{2^j\le n<M}r_\theta(n).
\]
\end{proof}

\begin{lemma}\label{IA:lem:TD1}
Let $\theta=a/q\in\mathcal Q$, let $d\ge1$, $L=2^d$, let $b$ be an odd integer with $0<b<L$, and
suppose that $\theta'=\theta+\frac{b}{qL}=\frac{aL+b}{qL}$ belongs to $\mathcal Q$ (with denominator
$qL$). Then for every set $I$ of at most $qL$ consecutive integers,
\[
\Big|\sum_{n\in I}r_{\theta'}(n)\Big|\le\frac{|I|\,b}{L}+q,\qquad\text{and consequently}\qquad
\mathcal D(\theta')\le q(b+1).
\]
\end{lemma}

\begin{proof}
Let $n_0=\min I$ and $K=\lfloor|I|/q\rfloor$. Split $I$ into the $K$ blocks
$\{n_0+kq+s:0\le s<q\}$, $0\le k<K$, and a remainder of fewer than $q$ integers, which contributes at
most $q/2$ in absolute value. Put $c=b/(2L)\in(0,\frac12)$. Since $a$ is odd,
$\frac q2\theta'=\frac a2+c\equiv\frac12+c\pmod1$, so for every $n$,
\[
r_{\theta'}(n)+r_{\theta'}(n+\tfrac q2)=\rho(x)+\rho(x+\tfrac12+c)=\rho(x)-\rho(x+c),\qquad
x=n\theta'-\tfrac16 .
\]
Within each block pair $s$ with $s+\frac q2$, $0\le s<\frac q2$. Fix $s$, and let
$x_k=(n_0+kq+s)\theta'-\frac16$. Since $q\theta'=a+2c$ with $a\in\Z$, we have
$\rho(x_k)-\rho(x_k+c)=\rho(y_k)-\rho(y_k+c)$ with $y_k=x_0+2ck$. This difference is $0$ unless the
interval $(y_k,y_k+c]$ contains a point of $\frac12\Z$, in which case it is $\pm1$. The intervals
$(y_k,y_k+c]$, $0\le k<K$, are pairwise disjoint and contained in an interval of length $2cK$, which
contains at most $4cK+1$ points of $\frac12\Z$. Summing over the $q/2$ values of $s$, the blocks
contribute at most $\frac q2(4cK+1)=\frac{qKb}{L}+\frac q2\le\frac{|I|b}{L}+\frac q2$. This proves
the first bound. By \cref{IA:lem:TD0}(i) applied to $\theta'$, every interval sum of $r_{\theta'}$
equals a sum over fewer than $qL$ consecutive integers, so $\mathcal D(\theta')\le qL\cdot b/L+q$.
\end{proof}

\begin{lemma}\label{IA:lem:TD2}
Let $\theta=a/q\in\mathcal Q$, $\theta'\ge\theta$ and $n\ge0$ with $n(\theta'-\theta)<1/q$. Then
$r_{\theta'}(n)=r_\theta(n)$.
\end{lemma}

\begin{proof}
$n\theta-\frac16$ lies in $\frac1q\Z$, and $\rho$ is constant on each interval
$[\frac kq,\frac{k+1}q)$ because its jumps lie in $\frac12\Z\subset\frac1q\Z$. The point
$n\theta'-\frac16$ lies in the same interval as $n\theta-\frac16$.
\end{proof}

The next lemma produces, from $\theta\in\mathcal Q$, a nearby $\theta'\in\mathcal Q$ with a much
larger gain $G(\theta')$ per dyadic block.

\begin{lemma}\label{IA:lem:TD3}
Let $\theta=a/q\in\mathcal Q$ with $q=3\cdot2^p$, and put $h=q-a\ge1$. Let $m\ge p$ with $m\equiv
p\pmod 2$, let $b=h2^m+1$ and let $R\ge1$ be the order of $2$ modulo $3b$. For $t\in\N$ put
$d=m+2Rt$, $L=2^d$ and $\theta'=\frac{aL+b}{qL}$. Then $0<b<L$, $\theta'\in\mathcal Q$ with
denominator $qL=3\cdot2^{p+d}$, $p+d$ is even, and
\[
G(\theta')=\Delta(\theta')\ \ge\ \frac{q2^m}{12}-h-q-2 .
\]
\end{lemma}

\begin{proof}
Since $2^R\equiv1\pmod{3b}$ we have $2^R>3b$, hence $L\ge2^{2R}>b$. Moreover
$2^d=2^m(2^R)^{2t}\equiv2^m\pmod{3b}$.
\begin{enumerate}[label=(\arabic*),leftmargin=*]
\item $hL+1\equiv h2^m+1=b\equiv0\pmod b$, so $U=(hL+1)/b$ is an integer.
\item $U-1=h(2^d-2^m)/b$. Here $(2^d-2^m)/b$ is an integer divisible by $3$ (as $3b\mid2^d-2^m$)
and by $2^m$ (as $b$ is odd). Hence $q=3\cdot2^p$ divides $U-1$; in particular $U\equiv1\pmod 6$, say
$U=6v+1$.
\item Put $a'=aL+b$ and $q'=qL$. Since $aU+h\equiv a+h=q\equiv0\pmod q$, the number
$\tau=(aU+h)/q$ is an integer, and
\[
a'U=aLU+bU=aLU+hL+1=L(aU+h)+1=\tau q'+1 .
\]
Hence $\gcd(a',q')=1$ and $\gcd(\tau,U)=1$. Also $0<a'<aL+hL=q'$, so $\theta'=a'/q'\in\mathcal Q$;
the exponent of $2$ in $q'$ is $p+d=p+m+2Rt$, which is even. Moreover
$\theta'=\frac{\tau}{U}+\frac{1}{Uq'}$.
\item Let $N=q'/3=2^{p+d}$, so that $\Delta(\theta')=\sum_{N\le n<2N}r_{\theta'}(n)$. For $n$ in this
range put $z=n/q'\in[\frac13,\frac23)$ and let $m'\in\{0,\dots,U-1\}$ be the residue of $n\tau$
modulo $U$. Then $n\theta'-\frac16\equiv\frac{m'+z}{U}-\frac16\pmod 1$ with $\frac{m'+z}U\in[0,1)$,
so $r_{\theta'}(n)=\frac12$ if and only if $\frac{m'+z}{U}\in[\frac16,\frac23)$, that is,
$m'+z\in[v+\frac16,4v+\frac23)$. Since $z\in[\frac13,\frac23)$, this holds if and only if
$v\le m'\le 4v$. Hence on any $U$ consecutive integers in $[N,2N)$, on which $m'$ takes every value
once (as $\gcd(\tau,U)=1$), there are $3v+1$ values $+\frac12$ and $3v$ values $-\frac12$, and the
sum is $\frac12$.
\item Split $[N,2N)$ into $\lfloor N/U\rfloor$ blocks of $U$ consecutive integers and a remainder of
fewer than $U\le hL+1\le q'$ integers. By \cref{IA:lem:TD1}, the remainder contributes at most
$Ub/L+q=h+\frac1L+q$ in absolute value. Hence
\[
\Delta(\theta')\ge\frac12\Big(\frac NU-1\Big)-h-1-q .
\]
Finally $\frac N{2U}=\frac{qLb}{6(hL+1)}\ge\frac{qb}{6(h+1)}\ge\frac{q2^m}{12}$, using
$b=h2^m+1\ge(h+1)2^{m-1}$. As $p+d$ is even, $G(\theta')=\Delta(\theta')$.\qedhere
\end{enumerate}
\end{proof}

\begin{lemma}\label{IA:lem:TD4}
Let $\theta=a/q\in\mathcal Q$ with $q=3\cdot2^p$, let $c\ge0$, and let $N_0,X_0\in\N$. Suppose that
$G(\theta)>c\log2$ and $S_\theta(M)\ge c\log M$ for all $M\ge N_0$. Let $m,b,R$ be as in
\cref{IA:lem:TD3}, chosen so that $\frac{q2^m}{12}-h-q-2\ge c\log 2$, and let $\theta'_t$ denote
the number $\theta'$ of \cref{IA:lem:TD3} for the parameter $t$. Then for all sufficiently large $t$:
\begin{enumerate}[label=\textup{(\roman*)}]
\item $r_{\theta'_t}(n)=r_\theta(n)$ for $0\le n<X_0$;
\item $S_{\theta'_t}(M)\ge c\log M$ for all $M\ge N_0$.
\end{enumerate}
\end{lemma}

\begin{proof}
Write $\theta'=\theta'_t$, $d=m+2Rt$, $P'=p+d$, and let $\ell=\lceil\log_2b\rceil+1$ and
$J=d-\ell$. For $0\le n<2^J$ we have $n(\theta'-\theta)=\frac{nb}{q2^d}<\frac{b2^{-\ell}}{q}\le
\frac1{2q}$, so $r_{\theta'}(n)=r_\theta(n)$ by \cref{IA:lem:TD2}. In particular (i) holds once
$2^J\ge X_0$, and $S_{\theta'}(M)=S_\theta(M)$ for $M\le2^J$.

Let $B_0=q(b+1)$; by \cref{IA:lem:TD1} (with $\theta'\in\mathcal Q$ by \cref{IA:lem:TD3}),
$\mathcal D(\theta')\le B_0$, and $B_0$ does not depend on $t$. By \cref{IA:lem:TD0}(iii) for
$\theta$,
\[
\Phi(t):=S_\theta(2^p)+(J-p)G(\theta)-(p+\ell)B_0-c(P'+1)\log2-B_0
= d\,\big(G(\theta)-c\log 2\big)+C_0,
\]
where $C_0$ does not depend on $t$. Since $G(\theta)>c\log2$ and $d\to\infty$ as $t\to\infty$, we may
take $t$ so large that $\Phi(t)\ge0$, $J\ge p$ and $2^J\ge\max(X_0,N_0)$. We check (ii) for this $t$.

If $N_0\le M\le 2^J$, then $S_{\theta'}(M)=S_\theta(M)\ge c\log M$.

If $2^J\le M\le 2^{P'}$, the interval $[2^J,M)$ meets at most $P'-J=p+\ell$ dyadic blocks, on each of
which $\varepsilon$ is constant; hence
$S_{\theta'}(M)\ge S_{\theta'}(2^J)-(p+\ell)B_0=S_\theta(2^J)-(p+\ell)B_0
\ge c(P'+1)\log2+B_0\ge c\log M$, using $S_\theta(2^J)=S_\theta(2^p)+(J-p)G(\theta)$ and
$\Phi(t)\ge0$. In particular $S_{\theta'}(2^{P'})\ge c(P'+1)\log2+B_0$.

If $M>2^{P'}$, let $2^j\le M<2^{j+1}$ with $j\ge P'$. By \cref{IA:lem:TD0}(iii) for $\theta'$ (whose
denominator is $3\cdot2^{P'}$) and $G(\theta')\ge c\log2$,
\[
S_{\theta'}(M)\ge S_{\theta'}(2^{P'})+(j-P')G(\theta')-B_0\ge c(P'+1)\log2+(j-P')c\log2=c(j+1)\log2
\ge c\log M .\qedhere
\]
\end{proof}

\begin{proof}[Proof of \cref{IA:prop:TD}]
Let $\theta_1=\frac1{12}$, so $q=12$, $p=2$. For $4\le n<8$ we have
$\frac n{12}-\frac16=\frac{n-2}{12}\in[0,\frac12)$, so $\Delta(\theta_1)=2$ and
$G(\theta_1)=2>2\log2$. By \cref{IA:lem:TD0}(iii), $S_{\theta_1}(M)\ge S_{\theta_1}(4)+2(j-2)-6$ for
$2^j\le M<2^{j+1}$, $j\ge2$, so there is $N_1$ with $S_{\theta_1}(M)\ge\log M$ for $M\ge N_1$.

Suppose that for some $k\ge1$ we have constructed $\theta_1,\dots,\theta_k$ with
$\theta_i=a_i/q_i\in\mathcal Q$, and $N_1\le\dots\le N_k$, such that
\begin{enumerate}[label=(H\arabic*),leftmargin=*]
\item\label{IA:H1} $G(\theta_k)>(k+1)\log2$, and
\item\label{IA:H2} $S_{\theta_k}(M)\ge i\log M$ for all $1\le i\le k$ and all $M\ge N_i$.
\end{enumerate}
(For $k=1$ this was just checked.) Choose $m\ge p_k$ with $m\equiv p_k\pmod2$ and
$\frac{q_k2^m}{12}-h_k-q_k-2>(k+2)\log2$, where $q_k=3\cdot2^{p_k}$, $h_k=q_k-a_k$. Apply
\cref{IA:lem:TD4} with $\theta=\theta_k$, $c=k$ and $N_0=X_0=N_k$; its hypotheses hold by \ref{IA:H1}
and \ref{IA:H2}. Taking $t$ large enough we obtain $\theta_{k+1}=\theta'_t\in\mathcal Q$ such that
$r_{\theta_{k+1}}=r_{\theta_k}$ on $[0,N_k)$, $S_{\theta_{k+1}}(M)\ge k\log M$ for $M\ge N_k$, and in
addition
\begin{equation}\label{IA:eq:eta}
\eta_k:=\theta_{k+1}-\theta_k<\min\Big(\frac1{8q_k^2},\frac{\eta_{k-1}}4\Big)\qquad(\eta_0:=1),
\end{equation}
which is possible because $\theta'_t-\theta_k=b/(q_k2^{m+2Rt})\to0$. By \cref{IA:lem:TD3},
$G(\theta_{k+1})>(k+2)\log2$. Property \ref{IA:H2} for $\theta_{k+1}$ and $i\le k$ holds: for
$N_i\le M\le N_k$ we have $S_{\theta_{k+1}}(M)=S_{\theta_k}(M)\ge i\log M$, and for $M\ge N_k$ we have
$S_{\theta_{k+1}}(M)\ge k\log M\ge i\log M$. For $i=k+1$, \cref{IA:lem:TD0}(iii) and
$G(\theta_{k+1})>(k+2)\log2$ give some $N_{k+1}\ge N_k$ with $S_{\theta_{k+1}}(M)\ge(k+1)\log M$ for
all $M\ge N_{k+1}$.

The sequence $(\theta_k)$ is increasing, and by~\eqref{IA:eq:eta} its limit $\alpha$ satisfies
$0<\alpha-\theta_k<\frac43\eta_k<\frac{1}{6q_k^2}$. Since $q_{k+1}=q_k2^{d}>q_k$, we have
$q_k\to\infty$. If $\alpha=c'/s$ were rational, then $\frac1{sq_k}\le|\alpha-\frac{a_k}{q_k}|
<\frac1{6q_k^2}$ for all $k$, which is impossible for large $k$; so $\alpha$ is irrational. For each
$n\ge1$ the point $n\alpha-\frac16$ is irrational, hence not in $\frac12\Z$, so $\rho$ is continuous
there and $r_{\theta_k}(n)\to r_\alpha(n)$. Consequently $S_{\theta_k}(M)\to S_\alpha(M)$ for each $M$.
Given $i$ and $M\ge N_i$, we have $S_{\theta_k}(M)\ge i\log M$ for all $k\ge i$ by \ref{IA:H2}, and
letting $k\to\infty$ gives $S_\alpha(M)\ge i\log M$.
\end{proof}
